## 四、核心模块设计

### 4.1 翻译引擎抽象层

```python
# core/translator.py
from abc import ABC, abstractmethod
from dataclasses import dataclass

@dataclass
class TranslationResult:
    source_text: str
    translated_text: str
    source_lang: str
    target_lang: str
    engine: str

class TranslatorEngine(ABC):
    @abstractmethod
    def translate(self, text: str, source_lang: str, target_lang: str) -> TranslationResult:
        pass
    
    @abstractmethod
    def detect_language(self, text: str) -> str:
        pass

class TranslatorManager:
    """翻译管理器，统一调度翻译引擎"""
    def __init__(self):
        self.engines = {}
        self.current_engine = "online"
    
    def register_engine(self, engine):
        self.engines[engine.engine_name] = engine
    
    def translate(self, text, source_lang="auto", target_lang="zh") -> TranslationResult:
        engine = self.engines.get(self.current_engine)
        return engine.translate(text, source_lang, target_lang)
```

### 4.3 配置管理

```python
# app/config.py
import json
from pathlib import Path

DEFAULT_CONFIG = {
    "engine": "online",
    "api_url": "",
    "api_key": "",
    "api_model": "gpt-3.5-turbo",
    "source_lang": "auto",
    "target_lang": "zh",
    "auto_copy": True,
    "minimize_to_tray": True,
    "launch_at_startup": False,
    "theme": "light",
    "hotkey": "Ctrl+Shift+M",
}

class Config:
    def __init__(self, path="data/config.json"):
        self.path = Path(path)
        self.data = DEFAULT_CONFIG.copy()
        self.load()
    
    def load(self):
        if self.path.exists():
            self.data.update(json.loads(self.path.read_text()))
    
    def save(self):
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(json.dumps(self.data, ensure_ascii=False, indent=2))
    
    def get(self, key, default=None):
        return self.data.get(key, default)
    
    def set(self, key, value):
        self.data[key] = value
        self.save()
```

### 6.2 JavaScript API调用

```javascript
// web/scripts/app.js
async function callApi(method, ...args) {
    if (window.pywebview && window.pywebview.api) {
        return await window.pywebview.api[method](...args);
    }
    throw new Error('pywebview API not available');
}

async function translate(text) {
    const result = await callApi('translate', text);
    if (result.success) {
        document.getElementById('translated-text').textContent = result.translation;
    }
}
```
